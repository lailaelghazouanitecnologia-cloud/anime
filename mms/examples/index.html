Quick Start

import { createLayout } from 'mms';

// Create layout instance
const layout = createLayout('#container', {
  children: '.item',
  duration: 500,
  ease: 'outExpo',
  enterFrom: { opacity: 0, scale: 0.8 },
  leaveTo: { opacity: 0, scale: 0.8 },
});

// Before making DOM changes
layout.record();

// Make your DOM changes...
container.appendChild(newElement);
item.remove();

// Animate to new layout
layout.animate();

Using update() shorthand

// Combines record() and animate()
layout.update(() => {
  // Make DOM changes inside callback
  items.sort((a, b) => a.order - b.order);
  renderItems();
});

Configuration Options

createLayout(root, {
  // Target children to animate
  children: '.item',       // CSS selector or '*' for all

  // Animation timing
  duration: 500,            // ms or function
  delay: 0,                 // ms or stagger()
  ease: 'outExpo',          // easing function

  // Enter animation (new elements)
  enterFrom: {
    opacity: 0,
    translateY: 20,
    scale: 0.9,
  },

  // Leave animation (removed elements)
  leaveTo: {
    opacity: 0,
    translateY: -20,
    scale: 0.9,
  },

  // Swap animation (elements changing position)
  swapAt: {
    opacity: 0.5,
    scale: 0.95,
  },

  // CSS properties to animate
  properties: ['backgroundColor', 'borderRadius'],

  // Callbacks
  onComplete: (timeline) => {},
});
